% POSICIONAMENTO DE FIGURAS
% O algoritmo padrão de floats do LaTeX reposiciona figuras livremente,
% fazendo-as aparecer longe do ponto de inserção no Markdown e, em
% casos extremos, invadir parágrafos.
%
% Solução: pacote `float` com \floatplacement{figure}{H}  força cada
% figura a ser renderizada exatamente onde foi declarada no fonte.
% O pacote `caption` garante compatibilidade com o estilo de legendas
% do abntex2 quando o posicionamento [H] está ativo.
% -------------------------------------------------------------------
\usepackage{float}
\usepackage{caption}
\floatplacement{figure}{H}

% -------------------------------------------------------------------
% RECUO DO PRIMEIRO PARÁGRAFO APÓS CAPÍTULO
%
% O \chapter do abntex2 chama \@afterindentfalse internamente, o que
% instrui o LaTeX a suprimir o recuo do primeiro parágrafo após cada
% título — independentemente do valor de \parindent e mesmo com o
% pacote indentfirst carregado, pois o abntex2 redefine \chapter de
% forma a contornar o hook do indentfirst.
%
% Solução: redefine \@afterindentfalse como alias de \@afterindenttrue
% no preâmbulo, tornando-o no-op permanente. Nenhum heading pode mais
% suprimir o recuo — o \parindent de 1.3 cm é sempre aplicado.
% -------------------------------------------------------------------
\makeatletter
\let\@afterindentfalse\@afterindenttrue
\makeatother
