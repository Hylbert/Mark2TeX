% Artigo em estilo APA 7ª edição (classe apa7)
% Referência: CTAN apa7 package – https://ctan.org/pkg/apa7
\documentclass[man]{apa7}

% inputenc and fontenc are pdfLaTeX-only; XeLaTeX handles UTF-8 natively
% and uses fontspec for font selection instead of T1/lmodern.
\usepackage{fontspec}
\usepackage[american]{babel}
\usepackage{csquotes}

% -------------------------------------------------------------------
% FONTE  —  controlada por flags booleanas geradas pelo build.sh
%
%   font-arial     → Liberation Sans     (métricas idênticas à Arial)
%   font-helvetica → Nimbus Sans         (Helvetica livre, URW)
%   font-times     → TeX Gyre Termes     (clone livre do Times New Roman)
%   font-ubuntu    → Ubuntu              (família clássica, estáticas disponíveis)
%
% Padrão (nenhuma flag): TeX Gyre Termes (APA recomenda Times New Roman)
% -------------------------------------------------------------------
\usepackage{fontspec}
$if(font-arial)$
  \setmainfont{Liberation Sans}
$else$$if(font-helvetica)$
  \setmainfont{Nimbus Sans}
$else$$if(font-ubuntu)$
  \setmainfont{Ubuntu}
$else$
  \setmainfont{TeX Gyre Termes}
$endif$$endif$$endif$

\usepackage{graphicx}
\usepackage{amsmath,amssymb}
\usepackage{hyperref}

$if(font-arial)$
\setmainfont{Arial}
$else$$if(font-helvetica)$
\setmainfont{Helvetica}
$else$$if(font-ubuntu)$
\setmainfont{Ubuntu}
$else$$if(font-times)$
\setmainfont{Times New Roman}
$endif$$endif$$endif$$endif$

\title{$title$}
\shorttitle{$if(shorttitle)$$shorttitle$$else$$title$$endif$}

$for(author)$
$if(author.name)$
\author{$author.name$}
$if(author.affiliation)$\affiliation{$author.affiliation$}$endif$
$else$
\author{$author$}
$endif$
$endfor$

$if(course)$\course{$course$}$endif$
$if(instructor)$\professor{$instructor$}$endif$
$if(institution)$\affiliation{$institution$}$endif$

\abstract{$if(abstract)$$abstract$$else$Resumo do artigo em até 250 palavras.$endif$}

$if(keywords)$\keywords{$keywords$}$endif$

\begin{document}
\maketitle

$body$

$if(bibliography)$
\bibliographystyle{apa7}
\bibliography{$bibliography$}
$endif$

\end{document}
